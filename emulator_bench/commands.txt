# DeepEnzyme emulator_bench commands (kcat only)
# Run from the DeepEnzyme submodule root: EMULaToR/baselines/DeepEnzyme
# PY=/home/adhil/.conda/envs/mldb/bin/python

# Defaults: --base_dir EMULaToR/data/processed/datasets/kinetic_dataset/kcat,
#           --work_dir EMULaToR/outputs/baselines/DeepEnzyme.
DATA=../../data/processed/datasets/kinetic_dataset/kcat
WORK=../../outputs/baselines/DeepEnzyme

# Prepare structures, cache features and train every default split for three seeds.
# Split groups: random_splits_grouped_sequence, random_splits_grouped_smiles, uniprot_time_splits,
# enzyme_sequence_splits/threshold_*, substrate_splits/threshold_* (thresholds are found on disk).
CUDA_VISIBLE_DEVICES=0 $PY emulator_bench/run_split_benchmarks.py \
  --base_dir $DATA --work_dir $WORK --seeds 666 777 888

# Run jobs in parallel on several GPUs.
$PY emulator_bench/run_split_benchmarks.py \
  --base_dir $DATA --work_dir $WORK --seeds 666 777 888 \
  --gpus 0 1 --runs_per_gpu 2

# Smoke test: one split, one seed, 100 rows per split, one epoch.
CUDA_VISIBLE_DEVICES=1 $PY emulator_bench/run_split_benchmarks.py \
  --base_dir $DATA --work_dir <scratch_dir> --split_groups random_splits_grouped_sequence \
  --seeds 666 --limit_rows 100 --iteration 1

# Aggregate mean and std over seeds.
$PY emulator_bench/aggregate_deepenzyme_results.py --results_dir $WORK/results
